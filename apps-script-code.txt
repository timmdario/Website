// Google Apps Script - In ein Google Sheet einfügen unter Erweiterungen → Apps Script
// Dann deployen als Web App (Zugriff: "Jeder")
// Nach Änderungen: Bereitstellen → Bereitstellungen verwalten → Stift-Symbol → Neue Version

// Tab mit den RSVP-Antworten (Spalten siehe unten). Falls nicht vorhanden, wird der erste Tab genutzt.
var RSVP_SHEET_NAME = 'Antworten';
// Tab mit den Gästen für den Briefumschlag. Spalten: Code | Vorname | Nachname | Nachricht | Anrede
// Anrede: 0 = Mann, 1 = Frau, 2 = Mehrere. Die Code-Spalte auf "Nur Text" formatieren.
var GUEST_SHEET_NAME = 'Gäste';
// Tab mit den Trauzeugen. Spalten: Brautpaar | Name | Telefon
// Die Kontakte werden nur an Gäste mit gültigem Code ausgegeben.
var TRAU_SHEET_NAME = 'Trauzeugen';
// Protokolliert ausschließlich erfolgreiche manuelle Logins auf der Startseite.
var LOGIN_LOG_SHEET_NAME = 'Login-Protokoll';
// Tab mit den Musikwünschen (wird automatisch angelegt). Spalten: Zeitstempel | Vorname | Nachname | Songtitel | Interpret | Anmerkung
var SONG_SHEET_NAME = 'Musikwünsche';

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.type === 'song') {
      return saveSongWish(data);
    }
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(RSVP_SHEET_NAME) || ss.getSheets()[0];

    // Spalten: Zeitstempel, Name, E-Mail, Anzahl Gäste, Namen der weiteren Gäste,
    //           Auf dem Gelände, Übernachtung Freitag, Übernachtung Samstag,
    //           Übernachtungs-Wünsche, Frühstück Sonntag, Ernährungswünsche, Nachricht
    sheet.appendRow([
      new Date(),
      data.name || '',
      data.email || '',
      data.guests || '',
      data.guest_names || '',
      data.overnight || '',
      data.overnight_friday || '',
      data.overnight_saturday || '',
      data.overnight_prefs || '',
      data.breakfast || '',
      data.dietary || '',
      data.message || ''
    ]);

    return jsonOutput({ success: true });

  } catch (error) {
    return jsonOutput({ success: false, error: error.toString() });
  }
}

function doGet(e) {
  var params = (e && e.parameter) || {};
  var code = String(params.code || '').trim();
  if (params.action === 'lookup') {
    return lookupGuest(code);
  }
  if (params.action === 'contacts') {
    return contactsFor(code);
  }
  if (params.action === 'loginLog') {
    return recordSuccessfulLogin(code);
  }
  return jsonOutput({ status: 'ok' });
}

// Sucht die Gästezeile zu einem 6-stelligen Code
function findGuestRow(code) {
  if (!/^\d{6}$/.test(code)) return null;
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(GUEST_SHEET_NAME);
  if (!sheet) return null;
  var rows = sheet.getDataRange().getValues();
  for (var i = 1; i < rows.length; i++) {
    var rowCode = String(rows[i][0]).trim();
    if (rowCode && rowCode.padStart(6, '0') === code) {
      return rows[i];
    }
  }
  return null;
}

function lookupGuest(code) {
  var row = findGuestRow(code);
  if (!row) {
    return jsonOutput({ found: false });
  }
  return jsonOutput({
    found: true,
    firstName: String(row[1] || '').trim(),
    lastName: String(row[2] || '').trim(),
    message: String(row[3] || ''),
    anrede: row[4] === '' ? '' : String(row[4])
  });
}

// Separater Endpunkt: protokolliert nur, wenn der Code gültig ist.
function recordSuccessfulLogin(code) {
  var guest = findGuestRow(code);
  if (!guest) return jsonOutput({ success: false });

  try {
    var lock = LockService.getScriptLock();
    lock.waitLock(10000);
    try {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      var sheet = ss.getSheetByName(LOGIN_LOG_SHEET_NAME);
      if (!sheet) {
        sheet = ss.insertSheet(LOGIN_LOG_SHEET_NAME);
        sheet.appendRow(['Zeitstempel', 'Name']);
      } else if (sheet.getLastRow() === 0) {
        sheet.appendRow(['Zeitstempel', 'Name']);
      }

      var name = [String(guest[1] || '').trim(), String(guest[2] || '').trim()]
        .filter(Boolean).join(' ');
      sheet.appendRow([new Date(), name]);
    } finally {
      lock.releaseLock();
    }
    return jsonOutput({ success: true });
  } catch (error) {
    console.error('Login-Protokoll:', error);
    return jsonOutput({ success: false });
  }
}

function contactsFor(code) {
  if (!findGuestRow(code)) {
    return jsonOutput({ found: false, contacts: [] });
  }
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(TRAU_SHEET_NAME);
  var contacts = [];
  if (sheet) {
    var rows = sheet.getDataRange().getValues();
    for (var i = 1; i < rows.length; i++) {
      var brautpaar = String(rows[i][0] || '').trim();
      var name = String(rows[i][1] || '').trim();
      var phone = String(rows[i][2] || '').trim();
      if (name && phone) {
        contacts.push({ brautpaar: brautpaar, name: name, phone: phone });
      }
    }
  }
  return jsonOutput({ found: true, contacts: contacts });
}

// Speichert einen Musikwunsch. Nur mit gültigem Gäste-Code, Name kommt aus dem Gäste-Tab
// Akzeptiert einen Wunsch (title/artist/note) oder mehrere (songs: [{title, artist, note}, ...])
function saveSongWish(data) {
  var guest = findGuestRow(String(data.code || '').trim());
  if (!guest) {
    return jsonOutput({ success: false, error: 'Ungültiger Code' });
  }
  var list = (data.songs && data.songs.length) ? data.songs : [{ title: data.title, artist: data.artist, note: data.note }];
  var rows = [];
  for (var i = 0; i < list.length; i++) {
    var title = String(list[i].title || '').trim();
    if (!title) continue;
    rows.push([
      new Date(),
      String(guest[1] || '').trim(),
      String(guest[2] || '').trim(),
      title,
      String(list[i].artist || '').trim(),
      String(list[i].note || '').trim()
    ]);
  }
  if (!rows.length) {
    return jsonOutput({ success: false, error: 'Songtitel fehlt' });
  }
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(SONG_SHEET_NAME) || ss.insertSheet(SONG_SHEET_NAME);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(['Zeitstempel', 'Vorname', 'Nachname', 'Songtitel', 'Interpret', 'Anmerkung']);
  }
  sheet.getRange(sheet.getLastRow() + 1, 1, rows.length, 6).setValues(rows);
  return jsonOutput({ success: true, saved: rows.length });
}

function jsonOutput(obj) {
  return ContentService.createTextOutput(
    JSON.stringify(obj)
  ).setMimeType(ContentService.MimeType.JSON);
}
